% Restrained academic layout based on the two supplied defense templates.
% The deck deliberately uses one accent colour, flat tables, and source figures.

\definecolor{IUBlue}{RGB}{10,91,160}
\definecolor{IUDeepBlue}{RGB}{6,55,105}
\definecolor{Ink}{RGB}{20,24,31}
\definecolor{MidGray}{RGB}{92,98,108}
\definecolor{RuleGray}{RGB}{175,181,190}
\definecolor{SoftGray}{RGB}{245,247,249}
\definecolor{TableBlue}{RGB}{232,240,247}

\setbeamercolor{normal text}{fg=Ink,bg=white}
\setbeamercolor{structure}{fg=IUBlue}
\setbeamercolor{frametitle}{fg=Ink,bg=white}
\setbeamercolor{itemize item}{fg=IUBlue}
\setbeamercolor{itemize subitem}{fg=IUBlue}
\setbeamercolor{enumerate item}{fg=IUBlue}
\setbeamercolor{block title}{fg=white,bg=IUBlue}
\setbeamercolor{block body}{fg=Ink,bg=SoftGray}

\setbeamerfont{title}{size=\LARGE,series=\bfseries}
\setbeamerfont{subtitle}{size=\large}
\setbeamerfont{frametitle}{size=\Large,series=\mdseries}
\setbeamerfont{footline}{size=\scriptsize}
\setbeamerfont{block title}{series=\bfseries}

\setbeamersize{text margin left=0.56cm,text margin right=0.56cm}
\setbeamertemplate{navigation symbols}{}
\setbeamertemplate{blocks}[default]
\setbeamertemplate{itemize item}{\raise0.15ex\hbox{\color{IUBlue}\small$\bullet$}}
\setbeamertemplate{itemize subitem}{\raise0.12ex\hbox{\color{IUBlue}\scriptsize$\blacktriangleright$}}

\setbeamertemplate{frametitle}{%
  \nointerlineskip
  \begin{beamercolorbox}[wd=\paperwidth,ht=0.80cm,dp=0.18cm,leftskip=0.56cm,rightskip=0.48cm]{frametitle}%
    \usebeamerfont{frametitle}\insertframetitle
    \hfill{\color{IUBlue}\bfseries\scriptsize INNOPOLIS UNIVERSITY}
  \end{beamercolorbox}%
  \nointerlineskip
  \begin{beamercolorbox}[wd=\paperwidth,ht=0.07cm,dp=0cm]{title rule}\end{beamercolorbox}%
}
\setbeamercolor{title rule}{bg=IUBlue}

\setbeamertemplate{footline}{%
  \leavevmode
  \hbox{%
    \begin{beamercolorbox}[wd=.37\paperwidth,ht=0.29cm,dp=0.12cm,leftskip=0.56cm]{footline}%
      \color{MidGray}\insertshortauthor
    \end{beamercolorbox}%
    \begin{beamercolorbox}[wd=.37\paperwidth,ht=0.29cm,dp=0.12cm,center]{footline}%
      \color{MidGray}Innopolis, 2026
    \end{beamercolorbox}%
    \begin{beamercolorbox}[wd=.26\paperwidth,ht=0.29cm,dp=0.12cm,rightskip=0.56cm plus 1fil]{footline}%
      \color{MidGray}\insertframenumber\,/\,\inserttotalframenumber
    \end{beamercolorbox}%
  }
}

\newcommand{\source}[1]{%
  \par\vspace{0.5mm}{\color{MidGray}\fontsize{5.8}{6.6}\selectfont Source: #1\par}%
}

\newcommand{\observation}[1]{%
  \par\vspace{1.0mm}%
  \noindent\color{IUDeepBlue}\rule{\textwidth}{0.45pt}\par\vspace{0.7mm}%
  {\small\color{Ink}\textbf{Result.} #1\par}%
}

\newcommand{\scope}[1]{%
  \par\vspace{0.8mm}%
  {\footnotesize\color{MidGray}\textbf{Scope:} #1\par}%
}

\newcommand{\tablehead}[1]{\cellcolor{TableBlue}\textbf{#1}}
