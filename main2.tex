\begin{frame}[t]{Relevance and research problem}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.42\textwidth}
      \centering
      \includegraphics[width=0.93\linewidth,height=2.20cm,keepaspectratio]{Testing-board.jpg}\\[-0.2ex]
      {\footnotesize Wire-harness assembly}

      \vspace{0.18cm}
      \includegraphics[width=0.93\linewidth,height=2.08cm,keepaspectratio]{threads.png}\\[-0.2ex]
      {\footnotesize Surgical-thread manipulation}
    \end{column}

    \begin{column}{0.56\textwidth}
      \small
      A \textbf{deformable linear object (DLO)} is a cable, wire, rope,
      hose, or thread whose complete shape changes during manipulation.

      \vspace{0.20cm}
      \begin{itemize}
        \item Distributed compliance makes the cable shape high-dimensional;
          the two endpoints do not determine it.
        \item Material and dynamic parameters are difficult to identify for
          each object.
        \item Local shape controllers may fail for large transformations,
          including a reversal of concavity.
        \item Collision-free endpoint paths or static cable shapes do not
          guarantee collision-free motion of every cable segment.
      \end{itemize}

      \vspace{0.10cm}
      \textbf{Research problem.} Control and route a planar DLO
      with two robots using its geometry, without prior identification of
      cable parameters (bending, stiffness, ..).
    \end{column}
  \end{columns}
  % \source{Dissertation, Introduction and Chapter 1.}
\end{frame}

    \begin{column}{0.68\textwidth}
      \footnotesize
      \textbf{Goal.} Develop mathematical models and effective numerical
      methods for dual-arm shape control and routing of DLOs under large planar
      deformations and workspace obstacles, without prior identification of
      dynamic or material parameters.

      \vspace{0.16cm}
      \textbf{Research objectives}
      \begin{enumerate}
        \item Reconstruct and track a marker-free planar cable from images.
        \item Generate length-feasible intermediary configurations, including
          opposite-concavity transformations.
        \item Model local robot--cable kinematics and calculate bounded
          dual-arm motion.
        \item Plan a cable route among static obstacles with valid
          swept motion and endpoint correspondence.
        \item Implement and verify the developed models and numerical methods.
      \end{enumerate}

      \vspace{0.02cm}
    \end{column}

\begin{frame}[t]{Problem formulation}
  \centering
  \footnotesize
  One ordered geometric state supports two complementary tasks:
  \(
    \bm p=[\bm p_1,\ldots,\bm p_N]^{\mathsf T}\in\R^{2N},
    \quad \|\bm p_{n+1}-\bm p_n\|\approx l_s.
  \)

\begin{frame}[t]{Virtual feature points (VFPs)}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.53\textwidth}
      \footnotesize
      \begin{enumerate}
        \item Detect the marked tool and identify the starting cable point.
        \item Segment the marker-free cable using edges and morphology.
        \item Apply Guo--Hall thinning to extract a one-pixel centerline.
        \item Place the center of a circular mask on the starting point.
        \item Record the forward intersection of the mask and centerline as a
          new feature point.
        \item Slide the mask to the new feature point.
        \item Repeat Steps 5 and 6 until the cable end is reached.
      \end{enumerate}
			\small
  \observation{VFPs reconstruct an ordered, approximately equidistant cable
  state from one top-view camera image.}

			\[
	  \bm p=[\bm p_1,\ldots,\bm p_N]^{\mathsf T},
    \qquad \|\bm p_{n+1}-\bm p_n\|\approx l_s .

        K=\left\lfloor\frac{\|\bm\Delta_\sigma\|}{\lambda}\right\rfloor+1,

        \bm p_\sigma^\kappa=\bm p_\sigma^{\kappa-1}+\bm\delta_\sigma,

% 8. Motion planning

\begin{frame}[t]{Motion planning}
  {\small\color{IUDeepBlue}\textbf{Aim:} Control the robots' end effectors to
  follow each intermediary shape.}
  \vspace{0.08cm}

  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.49\textwidth}
      \footnotesize
      {\color{IUDeepBlue}\textbf{Constrained shape control}\par}
      \setlength{\abovedisplayskip}{5pt}
      \setlength{\belowdisplayskip}{5pt}

      For each intermediary profile \(\bm p^\kappa\), solve
      \[
        \underset{\dot{\bm r}}{\operatorname{minimize}}
        \quad
        \left\|\bm p^\kappa-\bm p\right\|_2^2
      \]
      \[
        \text{subject to}\qquad
        \dot{\bm p}=\bm J\dot{\bm r},
        \qquad |\dot{\bm r}|\leq\bm\nu .
      \]

      The two endpoint influence matrices are assembled as
      \[
        \bm J_{2N\times6}=\operatorname{blkdiag}(\bm J^1,\bm J^2),
        \qquad
        \bm r_m=[r_{mx},r_{my},\theta_m]^{\mathsf T}.
      \]

      \begin{beamercolorbox}[wd=\linewidth,sep=0.10cm]{block body}
        \footnotesize
        Cable points are associated with the nearer endpoint in nominal
        cable-order distance. The velocity bound \(\bm\nu\) enforces smooth,
        feasible dual-arm motion.
      \end{beamercolorbox}

      \vspace{0.10cm}
      The controller uses measured cable geometry and robot poses; it does not
      require identified bending, stiffness, or damping parameters.
    \end{column}

    \begin{column}{0.49\textwidth}
      \centering
      {\footnotesize\color{IUDeepBlue}\textbf{Diminishing-rigidity Jacobian}\par}
      \includegraphics[width=0.64\linewidth,height=2.35cm,keepaspectratio]{rigidity_geometry.png}

      \footnotesize
      \setlength{\abovedisplayskip}{4pt}
      \setlength{\belowdisplayskip}{4pt}
      \[
        D_n^m=\eta_n^m l_s,
        \qquad
        d_n^m=\left\|\bm p_n-
        \begin{bmatrix}r_{mx}\\r_{my}\end{bmatrix}\right\|,
      \]
      \[
        \mu_n^m=e^{-|D_n^m-d_n^m|},
      \]
      \[
        \bm J_n^m=\mu_n^m
        \begin{bmatrix}
          1&0&-D_n^m\sin\theta_m\\
          0&1&D_n^m\cos\theta_m
        \end{bmatrix}.
      \]

      \begin{beamercolorbox}[wd=\linewidth,sep=0.08cm]{block body}
        \footnotesize
        \(d_n^m\approx D_n^m\): the cable is stretched and coupling is strong.\\
        \(d_n^m<D_n^m\): slack increases and endpoint influence diminishes.
      \end{beamercolorbox}
    \end{column}
  \end{columns}
\end{frame}

    \begin{column}{0.56\textwidth}
      {\footnotesize\color{IUDeepBlue}\textbf{Sampling and projection}\par}
      \vspace{0.04cm}

      \begin{beamercolorbox}[wd=\linewidth,sep=0.09cm]{block body}
        \footnotesize
        \textbf{1. Generate a candidate.}
        \(
          C^{\mathrm{raw}}\sim
          \{\text{goal-biased }(p_g),\ \text{tree-biased }(p_t),\
          \text{broad random }(1-p_g-p_t)\}.
        \)
      \end{beamercolorbox}

      \vspace{0.06cm}
      \begin{beamercolorbox}[wd=\linewidth,sep=0.09cm]{block body}
        \footnotesize
        \textbf{2. Bias toward the opposite tree.}
        Select \(C^{\mathrm{target}}\in V_{\mathrm{opp}}\), then
        \[
          C^b=C^{\mathrm{raw}}-\eta\nabla
          \bigl(U_{\mathrm{att}}+U_{\mathrm{rep}}\bigr)(C^{\mathrm{raw}}).
        \]
      \end{beamercolorbox}

      \vspace{0.06cm}
      \begin{beamercolorbox}[wd=\linewidth,sep=0.09cm]{block body}
        \footnotesize
        \textbf{3. Project to a feasible shape.}
        \[
          C^{\mathrm{ref}}=\arg\min_{C_i\in V_{\mathrm{act}}}d(C_i,C^b),
          \qquad
          C^{\mathrm{relax}}=\operatorname{Relax}(C^b,C^{\mathrm{ref}},\mathcal O).
        \]
        Accept only if \(C^{\mathrm{relax}}\in\Cfree\).
      \end{beamercolorbox}
      \begin{beamercolorbox}[wd=\textwidth,sep=0.09cm]{block body}
    \footnotesize
    \textbf{4. Select a parent and insert.}
    For both endpoint orders \(\sigma\in\{+1,-1\}\), retain only candidates satisfying
    \(d(C_i,(C^{\mathrm{relax}})^\sigma)\leq\epsilon\) and
    \(\Gamma(C_i,(C^{\mathrm{relax}})^\sigma)=1\). Reject if none remains;
    otherwise insert the best feasible edge into \(T_{\mathrm{act}}\).
  \end{beamercolorbox}
    \end{column}
  \end{columns}

% 16. Conclusions

\begin{frame}[t]{Conclusions}
  \small
  \textbf{Main results}
  \begin{enumerate}
    \item VFPs reconstruct an ordered point state of a marker-free planar cable
      from a single RGB stream.
    \item ALI generates length-feasible intermediary profiles for large
      deformations, including reversal of concavity.
    \item The diminishing-rigidity Jacobian and bounded optimization convert
      observed geometry into dual-arm motion without identified cable parameters.
    \item All eight physical shape-control trials were completed with mean
      pointwise error of 4.5--12.4 mm.
    \item GCR produced valid ordered routes in all 10,000 benchmark runs; all
      ten MuJoCo executions achieved RMSE of 1.60--4.66 mm.
  \end{enumerate}

  \vspace{0.10cm}
  \begin{beamercolorbox}[wd=\textwidth,sep=0.11cm]{block body}
    \footnotesize
    \textbf{Assumptions and limitations:} planar, quasi-static manipulation;
    visible cable centerline without self-intersection; known static obstacles;
    global routing validated computationally and in MuJoCo rather than on hardware.
  \end{beamercolorbox}

  \vspace{0.12cm}
  \footnotesize
  \textbf{Further work:} 3D perception and manipulation, explicit handling of
  occlusion and self-contact, and physical global-routing experiments.
\end{frame}

\begin{frame}[t]{Appendix: energy-minimizing DLO model}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.40\textwidth}
      \centering
      {\footnotesize\color{IUDeepBlue}\textbf{Discrete cable model}\par}
      \includegraphics[width=\linewidth,height=2.82cm,keepaspectratio]{mass_spring.png}

      \footnotesize
      \begin{itemize}
        \item \(N-1\) linear springs preserve segment length;
        \item \(N-2\) torsion springs penalize bending;
        \item the grippers impose the two endpoint positions.
      \end{itemize}
    \end{column}

    \begin{column}{0.58\textwidth}
      \footnotesize
      {\color{IUDeepBlue}\textbf{Elastic energy}\par}
      \setlength{\abovedisplayskip}{4pt}
      \setlength{\belowdisplayskip}{4pt}
      \[
        E_l=\sum_{n=2}^{N}\frac{k_l}{2}
        \left(\|\bm p_n-\bm p_{n-1}\|-l_s\right)^2,
      \]
      \[
        E_b=\sum_{n=2}^{N-1}\frac{k_b}{2}\gamma_n^2,
        \qquad E(\bm p)=E_l(\bm p)+E_b(\bm p).
      \]

      \vspace{0.08cm}
      {\color{IUDeepBlue}\textbf{Quasi-static equilibrium}\par}
      \[
        \begin{aligned}
          \underset{\bm p_{\mathrm{inner}}}{\operatorname{minimize}}
          \quad & E(\bm p) \\
          \text{subject to}\quad
          & \bm p_1=\bm p_1^{\mathrm{grip}},\qquad
            \bm p_N=\bm p_N^{\mathrm{grip}},
        \end{aligned}
      \]

      Here \(\bm p_{\mathrm{inner}}=[\bm p_2,\ldots,\bm p_{N-1}]^{\mathsf T}\).
      The minimizer is the cable shape used after each gripper update.
    \end{column}
  \end{columns}

  \vspace{0.12cm}
  \centering
  \begin{tikzpicture}[
    stage/.style={draw=IUBlue,rounded corners=2pt,fill=TableBlue,
