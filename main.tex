% 6. Formal problem setup

\begin{frame}[t]{Problem setup}
  \begin{beamercolorbox}[wd=\textwidth,sep=4pt,left]{claimbar}
    \small Two related problems are studied for the \textbf{complete cable shape},
    with distinct inputs, constraints, and outputs.
  \end{beamercolorbox}

  \vspace{0.22cm}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.49\textwidth}
      \begin{block}{Shape control}
        \footnotesize
        \textbf{Given:} observed current shape \(\bm p\) and prescribed target
        shape \(\bm t\).

        \vspace{0.12cm}
        \textbf{Determine:} bounded dual-arm motions and intermediary shapes
        \(\bm p^1,\ldots,\bm p^K\) such that
        \[
          \bm p \longrightarrow \bm p^1 \longrightarrow \cdots
          \longrightarrow \bm p^K=\bm t .
        \]

        \textbf{Requirements:} admissible deformation, nominal cable length,
        and convergence of the pointwise shape error.
      \end{block}
    \end{column}

    \begin{column}{0.49\textwidth}
      \begin{block}{Global routing}
        \footnotesize
        \textbf{Given:} start and goal configurations
        \(C_{\mathrm{start}},C_{\mathrm{goal}}\) and a static obstacle set
        \(\mathcal O\).

        \vspace{0.12cm}
        \textbf{Determine:} an ordered configuration sequence
        \[
          \mathcal P=\{C_0,\ldots,C_K\},\quad
          C_0=C_{\mathrm{start}},\ C_K=C_{\mathrm{goal}}.
        \]

        \textbf{Requirements:} \(C_k\in\mathcal C_{\mathrm{free}}\) and an
        admissible swept deformation between every \(C_k\) and \(C_{k+1}\).
      \end{block}
    \end{column}
  \end{columns}

  \vspace{0.12cm}
  \begin{beamercolorbox}[wd=\textwidth,sep=3.4pt,left]{claimbar}
    \scriptsize\textbf{Scope.} Planar quasi-static motion and dual-arm endpoint grasps
    are assumed. Shape control requires no identified stiffness, damping, or learned
    dynamics; global routing assumes a known static obstacle map.
  \end{beamercolorbox}

  \note{0:55. Separate the two formal problems. Shape control maps an observed current shape and prescribed target to bounded robot motions through intermediary shapes. Global routing maps prescribed start and goal configurations plus an obstacle map to a sequence whose states and swept transitions are admissible. No VFP contribution is claimed here; the state representation and its reconstruction are introduced on the next two slides.}
\end{frame}

\begin{frame}[t]{DLO model}
  \begin{beamercolorbox}[wd=\textwidth,sep=2.5pt,left]{claimbar}
    \scriptsize\textbf{Purpose.} The mass--spring model provides the numerical
    evaluation environment; it is not an input to the controller.
  \end{beamercolorbox}

  \vspace{0.10cm}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.45\textwidth}
      \begin{block}{Discrete mechanical representation}
        \centering
        \includegraphics[width=0.88\linewidth,height=1.90cm,keepaspectratio]{mass_spring.png}
        \vspace{0.02cm}
        \begin{itemize}\scriptsize
          \item \textbf{point masses:} discrete cable state;
          \item \textbf{linear springs:} segment-length response;
          \item \textbf{torsional springs:} elastic bending;
          \item \textbf{damping:} suppression of numerical oscillation.
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.53\textwidth}
      \begin{block}{Equations of motion}
        \scriptsize
        \vspace{-0.4ex}
        \[
          \mathbf M\ddot{\bm p}=\bm f_l+\bm f_b+\bm f_d,
        \]
        \vspace{-2.0ex}
        \[
          E_l=\sum_{n=1}^{N-1}\frac{k_l}{2}
          \bigl(\|\bm p_{n+1}-\bm p_n\|-l_s\bigr)^2,
        \]
        \vspace{-2.0ex}
        \[
          \bm f_l=-\nabla_{\bm p}E_l.
        \]
        \vspace{-1.2ex}
      \end{block}
      \vspace{-0.03cm}
      \begin{block}{Numerical realization}
        \scriptsize
        \begin{itemize}
          \item backward-Euler time integration;
          \item conjugate-gradient solution of each implicit update;
          \item repeated studies over cable length, discretization, target shape,
            and interpolation step.
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
